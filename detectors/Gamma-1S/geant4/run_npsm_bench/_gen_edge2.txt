def parse_csv(file_path):
    data = {}
    table = []
    try:
        with open(file_path, "r", encoding="utf-8") as f:
            for line in f:
                if line.startswith("#"):
                    continue
                if "," in line:
                    key, value = line.strip().split(",", 1)
                    # Приводим к числу ОДИН РАЗ при чтении — все потребители
                    # ниже получают уже числа. Точечная конвертация по списку
                    # ключей оставляла mean/sem строками и роняла критерии K1/K2.
                    if key in ("em_deex", "crystal_mm"):
                        data[key] = value            # заведомо строковые поля
                    else:
                        try:
                            data[key] = int(value)
                        except ValueError:
                            try:
                                data[key] = float(value)
                            except ValueError:
                                data[key] = value
                elif line.strip() == "n_compt,count_absorbed,count_all":
                    continue
                elif line.strip().startswith("0,"):
                    break
            for line in f:
                if "," in line and not line.startswith("#"):
                    parts = line.strip().split(",")
                    if len(parts) >= 3:
                        table.append([int(parts[0]), int(parts[1]), int(parts[2])])
    except Exception as e:
        print(f"ПРЕДУПРЕЖДЕНИЕ: Не удалось прочитать файл {file_path}: {e}")
        return None
    if "energy_keV" not in data or "mean_ncompt_absorbed" not in data:
        print(f"ПРЕДУПРЕЖДЕНИЕ: В файле {file_path} отсутствуют необходимые поля")
        return None
    data["table"] = table
    return data

def read_spectrum_table(file_path):
    """Читает таблицу спектра из CSV файла, начиная с строки bin_keV."""
    table = []
    try:
        with open(file_path, "r", encoding="utf-8") as f:
            for line in f:
                if line.strip() == "bin_keV,count_edep,count_light":
                    break
            for line in f:
                if "," in line and not line.startswith("#"):
                    parts = line.strip().split(",")
                    if len(parts) >= 3:
                        table.append([int(parts[0]), int(parts[1]), int(parts[2])])
    except Exception as e:
        print(f"ПРЕДУПРЕЖДЕНИЕ: Не удалось прочитать таблицу спектра из {file_path}: {e}")
        return None
    return table

def smooth(data, window=9):
    """Скользящее среднее по window бинам."""
    if len(data) < window:
        return data[:]
    smoothed = [0] * len(data)
    for i in range(len(data)):
        start = max(0, i - window // 2)
        end = min(len(data), i + window // 2 + 1)
        smoothed[i] = sum(data[start:end]) / (end - start)
    return smoothed

def find_peak_position(counts, start_bin=1):
    """Находит пик как бин с максимумом в заданном диапазоне."""
    max_count = 0
    peak_bin = -1
    for i in range(start_bin, len(counts)):
        if counts[i] > max_count:
            max_count = counts[i]
            peak_bin = i
    return peak_bin

def find_peak_position_centroid(counts, peak_bin):
    """Находит позицию пика как центр тяжести по бинам выше половины максимума."""
    if peak_bin < 0:
        return -1.0
    max_count = counts[peak_bin]
    threshold = max_count / 2.0
    total_weighted = 0.0
    total_weight = 0.0
    for i in range(peak_bin, len(counts)):
        if counts[i] >= threshold:
            total_weighted += i * counts[i]
            total_weight += counts[i]
        else:
            break
    if total_weight > 0:
        return total_weighted / total_weight
    return float(peak_bin)

def find_edge_position(smoothed_counts, edge_energy, plateau_start, plateau_end):
    """Находит положение края как точку, где сглаженный счет опускается ниже половины уровня плато."""
    # Уровень плато: медиана в области [0.45*E_edge, 0.85*E_edge]
    plateau_bins = []
    for i in range(len(smoothed_counts)):
        if plateau_start <= i <= plateau_end:
            plateau_bins.append(smoothed_counts[i])
    if len(plateau_bins) < 5:
        return None
    plateau_level = sorted(plateau_bins)[len(plateau_bins)//2]
    threshold = plateau_level / 2.0

    # Ищем первый бин, где счет ниже порога, предыдущий был выше
    for i in range(int(plateau_end) + 1, len(smoothed_counts)):
        if i > int(1.35 * edge_energy):
            break
        if smoothed_counts[i] < threshold:
            # Проверяем, что предыдущий был выше порога
            if i > 0 and smoothed_counts[i-1] >= threshold:
                # Линейная интерполяция между i-1 и i
                if smoothed_counts[i] == smoothed_counts[i-1]:
                    return float(i)
                x = (threshold - smoothed_counts[i-1]) / (smoothed_counts[i] - smoothed_counts[i-1])
                return float(i-1) + x
    return None

def calculate_edge_shift(file_path):
    """Вычисляет сдвиг комптоновского края для одного файла."""
    # Сначала читаем данные из файла
    data = parse_csv(file_path)
    if not data:
        print(f"Ошибка: не удалось прочитать файл {file_path}")
        return None

    energy_keV = data.get("energy_keV")
    npsm_enabled = data.get("npsm_enabled", 1)

    # Читаем таблицу спектра
    table = read_spectrum_table(file_path)
    if not table:
        print(f"Ошибка: не удалось прочитать таблицу спектра из файла {file_path}")
        return None

    bin_keV = [row[0] for row in table]
    count_edep = [row[1] for row in table]
    count_light = [row[2] for row in table]

    # Теоретический край
    E_edge = energy_keV * (2 * energy_keV / 511) / (1 + 2 * energy_keV / 511)

    # Калибровка: найти пик полного поглощения в области [0, 2*E, 1.05*E]
    peak_bin = find_peak_position(count_light, start_bin=1)
    if peak_bin < 0:
        print(f"Ошибка: не найден пик полного поглощения в файле {file_path}")
        return None

    # Уточнить положение как центр тяжести
    peak_pos = find_peak_position_centroid(count_light, peak_bin)
    if peak_pos < 0:
        print(f"Ошибка: не удалось определить позицию пика в файле {file_path}")
        return None

    # Коэффициент калибровки
    k = energy_keV / peak_pos

    # Сгладить кривую по свету
    smoothed_light = smooth(count_light, window=9)

    # Найти положение края по энергии (исходная шкала)
    edge_energy = E_edge
    plateau_start = int(0.45 * edge_energy)
    plateau_end = int(0.85 * edge_energy)
    edge_pos_energy = find_edge_position(smooth(count_edep, window=9), edge_energy, plateau_start, plateau_end)

    # Найти положение края по свету (калиброванная шкала)
    calibrated_light = [count_light[i] * k for i in range(len(count_light))]
    smoothed_calibrated = smooth(calibrated_light, window=9)
    edge_pos_light = find_edge_position(smoothed_calibrated, edge_energy, plateau_start, plateau_end)

    if edge_pos_energy is None or edge_pos_light is None:
        print(f"Ошибка: не удалось определить положение края в файле {file_path}")
        return None

    # Сдвиг
    shift = edge_pos_light - edge_pos_energy

    return {
        "energy_keV": energy_keV,
        "E_edge": E_edge,
        "edge_pos_energy": edge_pos_energy,
        "peak_pos": peak_pos,
        "k": k,
        "edge_pos_light": edge_pos_light,
        "shift": shift,
        "npsm_enabled": npsm_enabled
    }

def main():
    parser = argparse.ArgumentParser(description="Анализатор сдвига комптоновского края")
    parser.add_argument("--run", action="append", help="Файл CSV для анализа")
    parser.add_argument("--json", help="Выходной JSON файл")

    args = parser.parse_args()

    if not args.run:
        print("Ошибка: не указаны файлы для анализа (--run)")
        sys.exit(1)

    results = []
    for file_path in args.run:
        result = calculate_edge_shift(file_path)
        if result:
            results.append(result)

    if not results:
        print("Ошибка: не удалось обработать ни один файл")
        sys.exit(2)

    # Проверка критериев
    criteria_results = {}
    all_shifts = [r["shift"] for r in results]
    energies = [r["energy_keV"] for r in results]

    # К-1: край по ЭНЕРГИИ совпадает с теоретическим
    k1_pass = True
    for r in results:
        edge_pos_energy = r["edge_pos_energy"]
        E_edge = r["E_edge"]
        diff = abs(edge_pos_energy - E_edge)
        if diff > 0.08 * E_edge:
            k1_pass = False
            break

    criteria_results["K-1"] = {
        "threshold": "расхождение ≤ 8 % от E_edge",
        "fact": "все сходятся" if k1_pass else "есть расхождения",
        "result": "PASS" if k1_pass else "FAIL"
    }

    # К-2: знак сдвига отрицательный
    k2_pass = all(shift < 0 for shift in all_shifts)
    criteria_results["K-2"] = {
        "threshold": "сдвиг < 0 на всех поданных энергиях",
        "fact": "все отрицательны" if k2_pass else "есть положительные",
        "result": "PASS" if k2_pass else "FAIL"
    }

    # К-3: величина сдвига при 662 кэВ
    shift_662 = None
    for r in results:
        if r["energy_keV"] == 662:
            shift_662 = r["shift"]
            break
    k3_pass = False
    if shift_662 is not None:
        abs_shift = abs(shift_662)
        k3_pass = 12 <= abs_shift <= 28

    criteria_results["K-3"] = {
        "threshold": "12…28 кэВ по модулю",
        "fact": f"{abs_shift:.1f} кэВ" if shift_662 is not None else "не определено",
        "result": "PASS" if k3_pass else "FAIL"
    }

    # К-4: прогон с npsm_enabled = 0
    npsm_disabled_results = [r for r in results if r["npsm_enabled"] == 0]
    k4_pass = True
    if npsm_disabled_results:
        shift_disabled = npsm_disabled_results[0]["shift"]
        abs_shift_disabled = abs(shift_disabled)
        k4_pass = abs_shift_disabled <= 3

    criteria_results["K-4"] = {
        "threshold": "сдвиг не более 3 кэВ по модулю",
        "fact": f"{abs_shift_disabled:.1f} кэВ" if npsm_disabled_results else "не проверено (нет выключенного прогона)",
        "result": "PASS" if k4_pass else "FAIL"
    }

    # К-5: воспроизводимость: сдвиг растёт по модулю с энергией
    k5_pass = False
    if len(energies) >= 3:
        abs_shifts = [abs(shift) for shift in all_shifts]
        # Проверяем монотонность (возрастание)
        increasing = True
        for i in range(1, len(abs_shifts)):
            if abs_shifts[i] < abs_shifts[i-1]:
                increasing = False
                break
        k5_pass = increasing

    criteria_results["K-5"] = {
        "threshold": "монотонно, не менее чем на 3 точках",
        "fact": "возрастающий" if k5_pass else "не монотонный",
        "result": "PASS" if k5_pass else "FAIL"
    }

    # Вывод результатов
    print("Энергия, Теоретический край, Край по энергии, ППП в свете, Коэффициент k, Край по свету, Сдвиг")
    for r in results:
        print(f"{r['energy_keV']}, {r['E_edge']:.2f}, {r['edge_pos_energy']:.2f}, "
              f"{r['peak_pos']:.2f}, {r['k']:.4f}, {r['edge_pos_light']:.2f}, {r['shift']:.2f}")

    print("\nКРИТЕРИИ:")
    for key, value in criteria_results.items():
        print(f"{key}: {value['threshold']} | {value['fact']} | {value['result']}")

    # Требует толкования
    total_counts = sum([sum([row[1], row[2]]) for row in [r["table"] for r in results if "table" in r]])
    print("\nТРЕБУЕТ ТОЛКОВАНИЯ:")
    if total_counts < 200:
        print("Мало отсчётов в области края (< 200)")
    # Проверка на несколько минимумов производной
    print("Проверка на наличие нескольких сравнимых минимумов производной не реализована")

    # Итог
    final_shift = sum(all_shifts) / len(all_shifts)
    acceptance = "PASS" if all(c["result"] == "PASS" for c in criteria_results.values()) else "FAIL"
    print(f"\nSTAGE4_EDGE_SHIFT={final_shift:.2f} keV")
    print(f"STAGE4_ACCEPTANCE={acceptance}")

    # Сохранение в JSON
    if args.json:
        output_data = {
            "results": results,
            "criteria": criteria_results,
            "final_shift": final_shift,
            "acceptance": acceptance
        }
        with open(args.json, "w", encoding="utf-8") as f:
            json.dump(output_data, f, ensure_ascii=False, indent=2)

if __name__ == "__main__":
    main()
