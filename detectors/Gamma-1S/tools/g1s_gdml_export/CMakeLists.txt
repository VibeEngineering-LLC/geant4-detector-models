cmake_minimum_required(VERSION 3.16)
project(g1s_gdml_export CXX)
set(CMAKE_CXX_STANDARD 17)
if(NOT G4ROOT OR NOT XERCESC_ROOT)
  message(FATAL_ERROR "need -DG4ROOT (Geant4 with GDML, e.g. C:/geant4-gdml-vfix) and -DXERCESC_ROOT")
endif()
file(GLOB G4_IMPORT_LIBS "${G4ROOT}/lib/*.lib")
file(GLOB XERCESC_IMPORT_LIBS "${XERCESC_ROOT}/lib/*.lib")
set(GEOM_DIR "${CMAKE_CURRENT_SOURCE_DIR}/../../geometry")
add_executable(g1s_gdml_export main.cc "${GEOM_DIR}/G1SDetector.cc")
target_include_directories(g1s_gdml_export PRIVATE "${GEOM_DIR}" "${G4ROOT}/include/Geant4" "${XERCESC_ROOT}/include")
target_link_libraries(g1s_gdml_export PRIVATE ${G4_IMPORT_LIBS} ${XERCESC_IMPORT_LIBS})
target_compile_definitions(g1s_gdml_export PRIVATE G4LIB_BUILD_DLL WIN32 NOMINMAX _USE_MATH_DEFINES)
target_compile_options(g1s_gdml_export PRIVATE /EHsc /MD /bigobj /utf-8 /wd4005 /wd4267 /wd4244)
